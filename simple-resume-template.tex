\newcommand{\density}[1]{%
  \def\D{#1}%

  \ifnum\D=1
    \newgeometry{left=1.6cm,right=1.6cm,top=1.25cm,bottom=1.25cm}
  \fi
  \ifnum\D=2
    \newgeometry{left=1.45cm,right=1.45cm,top=1.15cm,bottom=1.15cm}
  \fi
  \ifnum\D=3
    \newgeometry{left=1.35cm,right=1.35cm,top=1.05cm,bottom=1.05cm}
  \fi

  \ifnum\D=1 \setstretch{1.06}\fi
  \ifnum\D=2 \setstretch{1.03}\fi
  \ifnum\D=3 \setstretch{0.99}\fi

  \ifnum\D=1 \titlespacing*{\section}{0pt}{10pt}{4pt}\fi
  \ifnum\D=2 \titlespacing*{\section}{0pt}{8pt}{3pt}\fi
  \ifnum\D=3 \titlespacing*{\section}{0pt}{6pt}{2pt}\fi

  \ifnum\D=1
    \renewcommand{\Section}[1]{\section*{##1}\vspace{-2pt}\hrule height 0.8pt \vspace{6pt}}
  \fi
  \ifnum\D=2
    \renewcommand{\Section}[1]{\section*{##1}\vspace{-3pt}\hrule height 0.8pt \vspace{5pt}}
  \fi
  \ifnum\D=3
    \renewcommand{\Section}[1]{\section*{##1}\vspace{-4pt}\hrule height 0.8pt \vspace{4pt}}
  \fi

  \ifnum\D=1
    \setlist[itemize]{
      leftmargin=1.25em,
      labelindent=0pt,
      labelwidth=0pt,
      labelsep=0.7em,
      label=\textbullet,
      topsep=2pt,
      itemsep=2pt,
      parsep=0pt,
      partopsep=0pt
    }
  \fi

  \ifnum\D=2
    \setlist[itemize]{
      leftmargin=1.25em,
      labelindent=0pt,
      labelwidth=0pt,
      labelsep=0.65em,
      label=\textbullet,
      topsep=1.5pt,
      itemsep=1.5pt,
      parsep=0pt,
      partopsep=0pt
    }
  \fi

  \ifnum\D=3
    \setlist[itemize]{
      leftmargin=1.25em,
      labelindent=0pt,
      labelwidth=0pt,
      labelsep=0.6em,
      label=\textbullet,
      topsep=1pt,
      itemsep=1pt,
      parsep=0pt,
      partopsep=0pt
    }
  \fi

  \ifnum\D=1 \renewcommand{\arraystretch}{1.08}\fi
  \ifnum\D=2 \renewcommand{\arraystretch}{1.03}\fi
  \ifnum\D=3 \renewcommand{\arraystretch}{0.98}\fi
}

  \fi

    \renewcommand{\BulletTextSize}{\small}
  \fi
  \ifnum\B=3
    \renewcommand{\BodyTextSize}{\small}
    \renewcommand{\BulletTextSize}{\footnotesize}
  \fi
}

\AtBeginEnvironment{itemize}{\BulletTextSize}
\AtEndEnvironment{itemize}{\BodyTextSize}
